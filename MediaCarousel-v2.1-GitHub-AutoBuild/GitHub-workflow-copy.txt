name: Build desktop apps

on:
  workflow_dispatch:
  push:
    branches: [main, master]

permissions:
  contents: read

jobs:
  build:
    name: ${{ matrix.label }}
    runs-on: ${{ matrix.runner }}
    timeout-minutes: 30
    strategy:
      fail-fast: false
      matrix:
        include:
          - runner: windows-2022
            label: Windows-x64
            arch: x64
          - runner: macos-15
            label: Mac-Apple-Silicon
            arch: arm64
          - runner: macos-15-intel
            label: Mac-Intel
            arch: x64
    steps:
      - uses: actions/checkout@v5
      - uses: actions/setup-python@v6
        with:
          python-version: '3.12'
          architecture: ${{ matrix.arch }}
          cache: pip
      - name: Install build dependencies
        run: python -m pip install -r requirements-build.txt
      - name: Build native app (includes model tests)
        run: python build.py
      - name: Verify bundled app startup
        run: python ci_verify.py
      - name: Archive Windows application folder
        if: runner.os == 'Windows'
        shell: pwsh
        run: |
          New-Item -ItemType Directory -Force packages
          Compress-Archive -Path dist/MediaCarousel -DestinationPath packages/MediaCarousel-Windows-x64.zip
      - name: Archive Mac application with executable permissions
        if: runner.os == 'macOS'
        env:
          BUILD_LABEL: ${{ matrix.label }}
        run: |
          mkdir -p packages
          ditto -c -k --sequesterRsrc --keepParent dist/MediaCarousel.app "packages/MediaCarousel-$BUILD_LABEL.zip"
      - name: Upload executable package
        uses: actions/upload-artifact@v7
        with:
          name: MediaCarousel-${{ matrix.label }}
          path: packages/*.zip
          if-no-files-found: error
          retention-days: 30
